cmake_minimum_required(VERSION 3.22)
project(PeerChatMessagingFoundation LANGUAGES C CXX)
add_library(peerchat_messaging SHARED peerchat_core.cpp)
target_compile_features(peerchat_messaging PUBLIC cxx_std_17)
target_compile_definitions(peerchat_messaging PUBLIC PEERCHAT_PORTABLE_TEXT)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(peerchat_messaging PRIVATE -Wall -Wextra -Werror)
endif()
target_include_directories(peerchat_messaging PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(peerchat_messaging PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(ANDROID)
    # The pinned existing crypto dependency only. No TLS, media or downloads.
    set(PEERCHAT_DEPENDENCY_ROOT "$ENV{PEERCHAT_DEPENDENCY_ROOT}" CACHE PATH "Shared dependency sources")
    if(NOT PEERCHAT_DEPENDENCY_ROOT)
        set(PEERCHAT_DEPENDENCY_ROOT "C:/Peerchatdep/deps")
    endif()
    execute_process(COMMAND git -c "safe.directory=${PEERCHAT_DEPENDENCY_ROOT}/mbedtls" -C "${PEERCHAT_DEPENDENCY_ROOT}/mbedtls" rev-parse HEAD OUTPUT_VARIABLE mbed_revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE mbed_status)
    if(NOT mbed_status EQUAL 0 OR NOT mbed_revision STREQUAL "068ff080b369adfac81509f9b57b2afabaf82dc5")
        message(FATAL_ERROR "Pinned MbedTLS checkout is required; no automatic download")
    endif()
    set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    add_subdirectory("${PEERCHAT_DEPENDENCY_ROOT}/mbedtls" mbedtls EXCLUDE_FROM_ALL)
    target_sources(peerchat_messaging PRIVATE ../storage.cpp ../group.cpp tls_android.cpp)
    option(PEERCHAT_STORAGE_TEST_HOOKS "Debug-only test-profile Storage proof" OFF)
    if(PEERCHAT_STORAGE_TEST_HOOKS)
        target_compile_definitions(peerchat_messaging PRIVATE PEERCHAT_STORAGE_TEST_HOOKS)
    endif()
    target_link_libraries(peerchat_messaging PRIVATE mbedcrypto)
    # Keep statically linked C++ runtime symbols private to this C ABI library.
    target_link_options(peerchat_messaging PRIVATE "-Wl,--exclude-libs,ALL")
endif()
# Explicit source list: shared Storage/Profile only on Android; no Node, Windows
# platform APIs/Schannel, audio, RTC, or ScreenMedia. Android byte-feed TLS uses
# the profile's opaque platform provider, never a replacement messaging engine.
option(PEERCHAT_PORTABLE_TESTS "Build the host ABI/format regression" ON)
if(PEERCHAT_PORTABLE_TESTS)
    enable_testing()
    add_executable(portable_core_test ../tests/portable_core_test.cpp)
    target_link_libraries(portable_core_test PRIVATE peerchat_messaging)
    target_compile_features(portable_core_test PRIVATE cxx_std_17)
    add_test(NAME portable_core COMMAND portable_core_test)
endif()
