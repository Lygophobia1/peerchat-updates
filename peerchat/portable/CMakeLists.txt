cmake_minimum_required(VERSION 3.22)
project(PeerChatMessagingFoundation LANGUAGES C CXX)
add_library(peerchat_messaging SHARED peerchat_core.cpp)
target_compile_features(peerchat_messaging PUBLIC cxx_std_17)
target_compile_definitions(peerchat_messaging PUBLIC PEERCHAT_PORTABLE_TEXT)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(peerchat_messaging PRIVATE -Wall -Wextra -Werror)
endif()
target_include_directories(peerchat_messaging PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(peerchat_messaging PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(ANDROID)
    # Keep statically linked C++ runtime symbols private to this C ABI library.
    target_link_options(peerchat_messaging PRIVATE "-Wl,--exclude-libs,ALL")
endif()
# Explicit source list: no Node, Windows storage/TLS, audio, RTC, or ScreenMedia.
# Missing providers are reported unavailable; no dummy crypto/storage is linked.
option(PEERCHAT_PORTABLE_TESTS "Build the host ABI/format regression" ON)
if(PEERCHAT_PORTABLE_TESTS)
    enable_testing()
    add_executable(portable_core_test ../tests/portable_core_test.cpp)
    target_link_libraries(portable_core_test PRIVATE peerchat_messaging)
    target_compile_features(portable_core_test PRIVATE cxx_std_17)
    add_test(NAME portable_core COMMAND portable_core_test)
endif()
